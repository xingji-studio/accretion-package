cmake_minimum_required(VERSION 3.20)
project(AccretionPackage VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AP_BUILD_PACKAGER "Build the host packaging CLI" ON)
option(AP_BUILD_WORKBENCH "Build the Qt packaging workbench when Qt is available" ON)
option(AP_REQUIRE_WORKBENCH "Fail when a requested graphical workbench cannot be built" OFF)
option(AP_BUILD_RUNTIMES "Build native installer templates" ON)
set(AP_INSTALLER_MODES "gui;tui" CACHE STRING "Installer frontends to build")
set(AP_XJ380_ADAPTER_SOURCE "" CACHE FILEPATH "Native XJ380 v1 adapter source")
option(AP_BUILD_TESTS "Enable integration tests" ON)

if(CMAKE_SYSTEM_NAME STREQUAL "XJ380")
  set(AP_PLATFORM xj380)
  set(AP_BUILD_PACKAGER OFF)
  set(AP_BUILD_WORKBENCH OFF)
  set(AP_BUILD_TESTS OFF)
elseif(WIN32)
  set(AP_PLATFORM windows)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(AP_PLATFORM linux)
else()
  message(FATAL_ERROR "Supported targets are Windows, Linux and XJ380")
endif()

add_library(ap_lz4 STATIC third_party/lz4/lz4.c)
target_include_directories(ap_lz4 PUBLIC third_party/lz4)
set(AP_CORE base json hash io package manifest context transaction engine integrations)
list(TRANSFORM AP_CORE PREPEND "src/core/")
list(TRANSFORM AP_CORE APPEND ".cpp")
add_library(ap_core STATIC ${AP_CORE} src/platform/memory.cpp src/platform/${AP_PLATFORM}.cpp)
target_include_directories(ap_core PUBLIC include)
target_link_libraries(ap_core PUBLIC ap_lz4)
if(MSVC)
  target_compile_options(ap_core PRIVATE /W4 /GR-)
else()
  target_compile_options(ap_core PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti -fno-builtin -ffunction-sections -fdata-sections)
  target_compile_options(ap_lz4 PRIVATE -ffunction-sections -fdata-sections)
endif()
if(AP_PLATFORM STREQUAL "windows")
  target_link_libraries(ap_core PUBLIC shell32 ole32 uuid advapi32 user32 gdi32)
elseif(AP_PLATFORM STREQUAL "xj380")
  enable_language(ASM)
  target_sources(ap_core PRIVATE src/platform/xj380_api.cpp src/platform/xj380_adapter.cpp)
  target_compile_definitions(ap_core PUBLIC AP_XJ380)
  target_compile_options(ap_lz4 PRIVATE -I${PROJECT_SOURCE_DIR}/include -include ap/codec_config.h)
endif()

if(AP_BUILD_RUNTIMES)
  add_library(ap_ui STATIC src/ui/model.cpp src/ui/tui.cpp src/ui/${AP_PLATFORM}_gui.cpp)
  target_link_libraries(ap_ui PUBLIC ap_core)
  if(AP_PLATFORM STREQUAL "windows")
    target_link_libraries(ap_ui PUBLIC comctl32)
  elseif(AP_PLATFORM STREQUAL "linux")
    target_link_libraries(ap_ui PUBLIC ${CMAKE_DL_LIBS})
  endif()
  foreach(mode IN LISTS AP_INSTALLER_MODES)
    if(NOT mode MATCHES "^(gui|tui)$")
      message(FATAL_ERROR "AP_INSTALLER_MODES accepts gui and tui")
    endif()
    if(WIN32 AND mode STREQUAL "gui")
      add_executable(ap_setup_${mode} WIN32 src/runtime/main.cpp)
    else()
      add_executable(ap_setup_${mode} src/runtime/main.cpp)
    endif()
    target_link_libraries(ap_setup_${mode} PRIVATE ap_ui)
    if(WIN32)
      target_sources(ap_setup_${mode} PRIVATE src/runtime/windows_resources.cpp)
    endif()
    if(mode STREQUAL "tui")
      target_compile_definitions(ap_setup_${mode} PRIVATE AP_DEFAULT_TUI)
    endif()
    if(AP_PLATFORM STREQUAL "xj380")
      target_sources(ap_setup_${mode} PRIVATE src/platform/xj380_crt.S)
      if(AP_XJ380_ADAPTER_SOURCE)
        target_sources(ap_setup_${mode} PRIVATE ${AP_XJ380_ADAPTER_SOURCE})
      endif()
      target_link_options(ap_setup_${mode} PRIVATE -nostdlib -Wl,-no-pie -Wl,--build-id=none -Wl,--gc-sections -Wl,-T,${PROJECT_SOURCE_DIR}/cmake/xj380.ld)
      set_target_properties(ap_setup_${mode} PROPERTIES SUFFIX ".elf")
    elseif(NOT MSVC)
      target_link_options(ap_setup_${mode} PRIVATE -Wl,--gc-sections)
    endif()
    set_target_properties(ap_setup_${mode} PROPERTIES OUTPUT_NAME ap-setup-${AP_PLATFORM}-x64-${mode} RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/ap-stubs")
    if(CMAKE_CONFIGURATION_TYPES)
      set_target_properties(ap_setup_${mode} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/ap-stubs")
    endif()
    install(TARGETS ap_setup_${mode} RUNTIME DESTINATION bin/ap-stubs)
  endforeach()
endif()

if(AP_BUILD_PACKAGER)
  add_library(ap_epf STATIC third_party/elf2epf/src/converter.cpp third_party/elf2epf/src/elf.cpp third_party/elf2epf/src/epf.cpp third_party/elf2epf/src/io.cpp)
  target_include_directories(ap_epf PUBLIC third_party/elf2epf/include)
  add_executable(apack src/builder/main.cpp src/builder/pe.cpp src/builder/epf.cpp)
  target_link_libraries(apack PRIVATE ap_core ap_epf)
  if(AP_PLATFORM STREQUAL "linux")
    set_target_properties(apack PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    target_link_options(apack PRIVATE -Wl,--disable-new-dtags)
  endif()
  install(TARGETS apack RUNTIME DESTINATION bin)
endif()
set(AP_FRONTEND_KIND "tui")
set(AP_QT_CORE "")
set(AP_QMAKE "")
if(AP_BUILD_WORKBENCH)
  find_package(Qt6 COMPONENTS Widgets QUIET)
  if(Qt6_FOUND)
    add_executable(apack-workbench src/builder/workbench.cpp)
    target_link_libraries(apack-workbench PRIVATE Qt6::Widgets)
    set_property(TARGET apack-workbench PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    if(AP_PLATFORM STREQUAL "linux")
      set_target_properties(apack-workbench PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
      target_link_options(apack-workbench PRIVATE -Wl,--disable-new-dtags)
    endif()
    set(AP_FRONTEND_KIND "gui")
    set(AP_QT_CORE "$<TARGET_FILE:Qt6::Core>")
    if(TARGET Qt6::qmake)
      get_target_property(AP_QMAKE Qt6::qmake IMPORTED_LOCATION)
    endif()
    install(TARGETS apack-workbench RUNTIME DESTINATION bin)
  else()
    if(AP_REQUIRE_WORKBENCH)
      message(FATAL_ERROR "Qt 6 Widgets development files are missing. Set AP_QT_PREFIX/CMAKE_PREFIX_PATH to your Qt SDK, or select AP_FRONTEND=tui.")
    endif()
    message(STATUS "Qt 6 Widgets not installed: CLI/TUI builder and native GUI/TUI installers remain enabled")
  endif()
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/ap-build-info-$<CONFIG>.json" CONTENT
"{\"version\":\"${PROJECT_VERSION}\",\"platform\":\"${AP_PLATFORM}\",\"frontend\":\"${AP_FRONTEND_KIND}\",\"qt_core\":\"${AP_QT_CORE}\",\"qmake\":\"${AP_QMAKE}\"}\n")
if(AP_BUILD_TESTS)
  include(CTest)
  if(TARGET apack-workbench)
    add_test(NAME workbench-smoke COMMAND apack-workbench --smoke-test)
    set_tests_properties(workbench-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  add_executable(ap-core-tests tests/unit.cpp)
  target_link_libraries(ap-core-tests PRIVATE ap_core)
  add_test(NAME core COMMAND ap-core-tests)
  add_executable(ap-xapi-tests tests/xapi_abi.cpp src/platform/xj380_api.cpp)
  target_include_directories(ap-xapi-tests PRIVATE include)
  target_compile_definitions(ap-xapi-tests PRIVATE AP_XAPI_TEST)
  add_test(NAME xapi-abi COMMAND ap-xapi-tests)
  add_executable(ap-xj-adapter-tests tests/xj_adapter.cpp src/platform/xj380_adapter.cpp src/core/base.cpp src/core/json.cpp src/platform/memory.cpp)
  target_include_directories(ap-xj-adapter-tests PRIVATE include)
  target_compile_definitions(ap-xj-adapter-tests PRIVATE AP_XJ_ADAPTER_TEST)
  if(WIN32)
    target_link_libraries(ap-xj-adapter-tests PRIVATE user32)
  endif()
  add_test(NAME xj-adapter-contract COMMAND ap-xj-adapter-tests)
  if(WIN32)
    add_executable(ap-windows-registry-tests tests/windows_registry.cpp)
    target_link_libraries(ap-windows-registry-tests PRIVATE ap_core)
    add_test(NAME windows-registry COMMAND ap-windows-registry-tests)
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME build-tools COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/build_tools.py)
  endif()
  if(Python3_FOUND AND AP_BUILD_PACKAGER AND AP_BUILD_RUNTIMES)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/integration.py --build-dir $<TARGET_FILE_DIR:apack>)
  endif()
endif()
